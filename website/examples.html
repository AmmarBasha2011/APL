متتالية فيبوناتشي

دوال وحلقات وتكرار

دالة فيب(ن):
    لو ن < 2:
        ارجع ن
    ارجع فيب(ن - 1) + فيب(ن - 2)

دالة فيب_تكراري(عدد):
    لو عدد < 2:
        ارجع عدد
    السابق = 0
    الحالي = 1
    كرارة = 2
    طالما كرارة <= عدد:
        التالي = السابق + الحالي
        السابق = الحالي
        الحالي = التالي
        كرارة = كرارة + 1
    ارجع الحالي

اطبع "=== متتالية فيبوناتشي ==="
اطبع "أول 8 حدود:"
المتغير الحدود = 0
طالما الحدود < 8:
    اطبع f"  فيب({الحدود}) = {فيب(الحدود)}"
    الحدود = الحدود + 1
اطبع ""
اطبع "فيب(10) تكراري: " + نص(فيب_تكراري(10))
اطبع "فيب(20) تكراري: " + نص(فيب_تكراري(20))
اطبع "فيب(50) تكراري: " + نص(فيب_تكراري(50))

دوال وشروط وقوائم

# مثال 12 — فحص الأعداد الأولية
# Prime Number Checker
# Demonstrates: sqrt, modulo, user functions, early exit
#
# Run:  python apl.py examples/12-prime-check.apl

دالة أولي(عدد):
    لو عدد < 2:
        ارجع خطأ
    لو عدد == 2:
        ارجع صواب
    لو عدد % 2 == 0:
        ارجع خطأ
    القاسم = 3
    طالما القاسم * القاسم <= عدد:
        لو عدد % القاسم == 0:
            ارجع خطأ
        القاسم = القاسم + 2
    ارجع صواب

دالة أوليات_حتى(الحد):
    النتيجة = []
    رقم = 2
    طالما رقم <= الحد:
        لو أولي(رقم):
            النتيجة.ألحق(رقم)
        رقم = رقم + 1
    ارجع النتيجة

اطبع "=== فحص الأعداد الأولية ==="
المتغير قائمة = أوليات_حتى(50)
اطبع "الأعداد الأولية حتى 50:"
اطبع "  " + نص(قائمة)
اطبع ""
اطبع "عددها: " + نص(طول(قائمة))
اطبع ""
اطبع "اختبار:"
لكل رقم في [1, 2, 7, 9, 13, 21, 97, 100]:
    اطبع f"  {رقم} أولي؟ {أولي(رقم)}"

الملف: examples/12-prime-check.apl

خوارزميات

# مثال 6 — خوارزميات البحث والترتيب
# Searching & Sorting Algorithms
# Demonstrates: custom functions, nested while-loops, list copy, mutation
#
# Run:  python apl.py examples/06-search-sort.apl

دالة فقاعة(المدخل):
    الناتج = المدخل.نسخ()
    n = طول(الناتج)
    i = 0
    طالما i < n:
        j = 0
        طالما j < n - 1 - i:
            إذا الناتج[j] > الناتج[j + 1]:
                مؤقت = الناتج[j]
                الناتج[j] = الناتج[j + 1]
                الناتج[j + 1] = مؤقت
            j = j + 1
        i = i + 1
    ارجع الناتج

دالة بحث(القائمة, الهدف):
    رقم = 0
    طالما رقم < طول(القائمة):
        إذا القائمة[رقم] == الهدف:
            ارجع رقم
        رقم = رقم + 1
    ارجع -1

المتغير البيانات = [64, 25, 12, 22, 11, 90, 33]

اطبع "=== البحث والترتيب ==="
اطبع "الأصل: " + نص(البيانات)
المتغير المرتب = فقاعة(البيانات)
اطبع "مرتب: " + نص(المرتب)
اطبع ""
اطبع "بحث عن 22: الفهرس " + نص(بحث(البيانات, 22))
اطبع "بحث عن 99: الفهرس " + نص(بحث(البيانات, 99))
اطبع "أكبر عنصر: " + نص(المرتب[طول(المرتب) - 1])
اطبع "أصغر عنصر: " + نص(المرتب[0])

# مثال 1 — آلة حاسبة تفاعلية
# Interactive Calculator
# Demonstrates: input, arithmetic, multi-level conditionals, printing
#
# Run:  echo "1" | python apl.py examples/01-calculator.apl

المتغير العملية = ادخل("العملية (1=جمع، 2=طرح، 3=ضرب، 4=قسمة): ")

بذرة(7)
المتغير أ = عشوائي_عدد(10, 99)
المتغير ب = عشوائي_عدد(10, 99)

اطبع ""
اطبع "=== آلة حاسبة APL ==="
اطبع "الأول: " + نص(أ)
اطبع "الثاني: " + نص(ب)

إذا العملية == "1":
    اطبع "النتيجة (جمع): " + نص(أ + ب)
سإذا العملية == "2":
    اطبع "النتيجة (طرح): " + نص(أ - ب)
سإذا العملية == "3":
    اطبع "النتيجة (ضرب): " + نص(أ * ب)
سإذا العملية == "4":
    إذا ب == 0:
        اطبع "خطأ: القسمة على صفر"
    والا:
        اطبع "النتيجة (قسمة): " + نص(أ / ب)
والا:
    اطبع "عملية غير معروفة"
