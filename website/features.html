1. إمسك مثل — التقاط الاستثناءات

دالة_قراءة(ملف):
    حاول:
        النص = اقرأ(ملف)
        ارجع "نجح: " + النص
    إمسك مثل خطأ:
        اطبع "تعذّر القراءة: " + نص(خطأ)
        ارجع "فشل"
دالة_مقسوم(أ, ب):
    حاول:
        ارجع أ / ب
    إمسك مثل خطأ:
        ارجع 0

كان مكسورًا قبل هذا الإصلاح: خطأ كلمة أساسية تعني False، فكان ينتج:

except Exception as False:   # SyntaxError

الآن يعمل.

2. حالة / قيمة — match/case

الصيغة الأولى: قيمة

دالة_وصف(ن):
    حالة ن:
        قيمة 0:
            ارجع "صفر"
        قيمة 1:
            ارجع "واحد"
        قيمة _:
            ارجع "غير معروف"

الصيغة الثانية: حالة (نفس المعنى)

دالة_وصف(ن):
    حالة ن:
        حالة 0:
            ارجع "صفر"
        حالة _:
            ارجع "غير معروف"

حماية (guard)

دالة_تصنيف(ن):
    حالة ن:
        قيمة ن لو ن < 0:
            ارجع "سالب"
        قيمة ن لو ن == 0:
            ارجع "صفر"
        قيمة ن لو ن < 10:
            ارجع "صغير"
        قيمة _:
            ارجع "كبير"

بدائل (alternatives)

دالة_يوم(ن):
    حالة ن:
        قيمة (6 | 7):
            ارجع "عطلة"
        قيمة _:
            ارجع "عمل"

مقارنة نصوص

دالة_نوع_المادة(م):
    حالة م:
        قيمة "رياضيات":
            ارجع "علمية"
        قيمة "تاريخ":
            ارجع "نظري"
        قيمة _:
            ارجع "معروف"

3. lambda

مهمة

المتغير مربع = (س) => س * س
اطبع مربع(5)     # 25

داخل الاستدعاء

المتغير الأرقام = [1, 2, 3, 4]
اطبع قائمة(خريطة((س) => س * س, الأرقام))    # [1, 4, 9, 16]
اطبع قائمة(فلترة_قائمة((س) => س % 2 == 0, الأرقام))  # [2, 4]

كـ callback

المتغير الكلمات = ["تفاحة", "موز", "كرز", "بطيخ"]
المتغير مرتبة = فرز_قائمة(الكلمات, ترتيب_مفتاح = (كلمة) => طول(كلمة))
اطبع مرتبة

4. الوراثة

صنف حيوان:
    دالة_إنشاء(ذات, اسم, صوت):
        ذات.اسم = اسم
        ذات.صوت = صوت

    دالة_تحدث(ذات):
        ارجع ذات.اسم + " تقول " + ذات.صوت

    دالة_وصف(ذات):
        ارجع "حيوان: " + ذات.اسم


صنف كلب(حيوان):
    دالة_تحدث(ذات):
        ارجع ذات.اسم + " ينبح"

    دالة_وصف(ذات):
        ارجع "كلب: " + ذات.اسم


المتغير د = كلب("بودي", "woof")
اطبع د.تحدث()     # بودي ينبح
اطبع د.وصف()      # كلب: بودي

صنف داخل صنف

صنف حاوية:
    صنف عنصر:
        دالة_إنشاء(ذات, قيمة):
            ذات.قيمة = قيمة

        دالة_نص(ذات):
            ارجع f"القيمة: {ذات.قيمة}"

المتغير ك = حاوية.عنصر(42)
اطبع ك.نص()

5. صيغة دالة_Prefix — متى تستخدمها

المكانالصيغةالنتيجة في Python
دالة علويةدالة_فيب(ن):def فيب(ن): (الاستدعاء الذاتي يعمل)
method في صنفدالة_م(ذات):def م(self):
constructorدالة_إنشاء(ذات, ...):def __init__(self, ...):
دالة متداخلةدالة_غلاف(س):غلاف = ... (بدون بادئة)

القاعدة: دالة_ على المستوى العلوي مطلوبة ليستدعي الدالة نفسها؛ داخل الصنف يكفي اسم عادي.
