# مرجع اللغة — APL

## المتغيرات

```apl
المتغير اسم = "عمار"
المتغير عمر = 25
المتغير س = صحيح("42")
اطبع "الاسم:", اسم, "| العمر:", عمر, "| الرقم:", س
```

## الطباعة والإدخال

```apl
المتغير اسم = "عمار"

اطبع "مرحبا"                     # print("مرحبا")
اطبع "الاسم:", اسم              # print("الاسم:", اسم)
اطبع_بدون "نص"                   # print("نص", end="")
اطبع ""

# ادخل("سؤال") يقرأ من لوحة المفاتيح:
# المتغير اسم = ادخل("ما اسمك؟ ")
```

## الشروط

```apl
المتغير عمر = 20

لو عمر >= 18:
    اطبع "بالغ"
الا لو عمر > 12:
    اطبع "مراهق"
الا:
    اطبع "طفل"

# شرط ثلاثي
المتغير ن = 1 إذا صواب والا 0
اطبع "الشرط الثلاثي:", ن
```

## الحلقات

```apl
المتغير س = 0
طالما س < 3:
    اطبع "طالما:", س
    س = س + 1

لكل i في نطاق(3):
    اطبع "لكل:", i

المتغير عداد = 0
أبدا:
    عداد = عداد + 1
    لو عداد >= 3:
        توقف
اطبع "أبدا توقفت بعد:", عداد
```

## List Comprehensions

```apl
# [تعبير لكل عنصر في تسلسل]
المتغير مربعات = [س*س لكل س في نطاق(1, 6)]
اطبع مربعات

# مع شرط
المتغير زوجي = [س لكل س في نطاق(1, 11) لو س % 2 == 0]
اطبع زوجي
```

## الدوال

```apl
دالة جمع(أ, ب):
    ارجع أ + ب

المتغير ضعف = سهم س: س * 2         # lambda

# Type Hints
دالة جمع(أ: صحيح, ب: صحيح) -> صحيح:
    ارجع أ + ب
```

## الفئات (Classes)

```apl
صنف شخص:
    دالة_إنشاء(ذات, اسم):
        ذات.اسم = اسم
    دالة سلام(ذات):
        اطبع "مرحبا", ذات.اسم

صنف طالب(شخص):
    تمرير

المتغير ش = شخص("علي")
ش.سلام()

المتغير ط = طالب("سارة")
ط.سلام()
```

## معالجة الأخطاء

```apl
حاول:
    المتغير س = صحيح("abc")
إمسك:
    اطبع "رقم غير صحيح"

```

## الملفات

```apl
# المقتطف مستقل: ينشئ ملفاته وينظّفها
اكتب("أ.txt", "محتوى\n")

مع فتح("أ.txt", "r") مثل ملف:
    اطبع اقرأ(ملف)

نسخ ملف "أ.txt" إلى "ب.txt"
اطبع "بعد النسخ:", موجود("ب.txt")

انشئ مجلد "جديد_مجلد"
اطبع "المجلد:", هو_مجلد("جديد_مجلد")

احذف ملف "ب.txt"
احذف ملف "أ.txt"
احذف_مجلد("جديد_مجلد")
```

## الوقت والتاريخ

| العربي | Python | الوصف |
|--------|--------|-------|
| `وقت()` | `time.time()` | وقت بالثواني |
| `وقت_نومي(ث)` | `time.sleep(ث)` | انتظار (ثواني) |
| `وقت_نص()` | `time.ctime()` | وقت كنص |

| `اليوم()` | `datetime.date.today()` | التاريخ الحالي |
| `سنة_الآن` | `datetime.datetime.now().year` | السنة (بدون أقواس) |
| `شهر_الآن` | `datetime.datetime.now().month` | الشهر (بدون أقواس) |
| `يوم_الآن` | `datetime.datetime.now().day` | اليوم (بدون أقواس) |
| `ساعة_الآن` | `datetime.datetime.now().hour` | الساعة (بدون أقواس) |
| `دقيقة_الآن` | `datetime.datetime.now().minute` | الدقيقة (بدون أقواس) |
| `ثانية_الآن` | `datetime.datetime.now().second` | الثانية (بدون أقواس) |

## المزخرفات (Decorators)

```apl
صنف دائرة:
    دالة_إنشاء(ذات, نصف):
        ذات._نصف = نصف

    خاصية نصف_القطر:
        ارجع ذات._نصف

    محدد نصف_القطر:
        ذات._نصف = قيمة

المتغير د = دائرة(5)
اطبع "نصف القطر:", د.نصف_القطر
د.نصف_القطر = 7
اطبع "بعد التعديل:", د.نصف_القطر
```
