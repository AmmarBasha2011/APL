`xml.etree.ElementTree` القياسية في Python (بدون أي مكتبة خارجية).

## مثال

```apl
المتغير المستند = "<مكتبة><كتاب نوع='علمي'><عنوان>فيزياء</عنوان></كتاب></مكتبة>"
المتغير الجذر = إكس_إم_إل_من_نص(المستند)

اطبع الجذر.اسم_العنصر                                  # مكتبة
لكل كتاب في الجذر.كرر("كتاب"):
    اطبع كتاب.اقرأ_خاصية("نوع") + ": " + كتاب.ابحث_نص("عنوان")

المتغير نسخة = انسخ_العنصر(الجذر)
عين_خاصية(نسخة, "تاريخ", "2026")
احفظ_إكس_إم_إل(نسخة, "مكتبة-نسخة.xml")
اطبع إكس_إم_إل_إلى_نص(نسخة)
```
